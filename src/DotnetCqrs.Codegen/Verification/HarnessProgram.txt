// Fixed, reusable scenario-verification harness -- copied verbatim into a scratch
// project alongside generated deciders/projections by ScenarioVerifier (never
// generated per-document). Reads a HarnessInput JSON file (args[0]), runs every
// scenario against the real, compiled generated code (in-memory SQLite, no live
// state touched), and writes a JSON array of ScenarioResult to args[1].
//
// This is dotnetcqrs's answer to pocketcqrs's "dry-run the JS source in a sandboxed
// VM" (functions.DryRunDecide/DryRunProjectionOver): there is no equivalent to
// evaluating source text without compiling it for C#, so verification here means
// actually compiling this harness together with the generated code once (a real
// `dotnet build`/`dotnet run`, same infrastructure Milestone 3's own tests proved
// out) and driving every scenario through the resulting assembly via reflection.

using System.Data.Common;
using System.Reflection;
using System.Text.Json;
using DotnetCqrs.Codegen.Generation;
using DotnetCqrs.Deciders;
using DotnetCqrs.EventStore;
using DotnetCqrs.Projections;
using DotnetCqrs.ReadModels;

var input = JsonSerializer.Deserialize<HarnessInput>(await File.ReadAllTextAsync(args[0]))!;
var results = new List<ScenarioResultDto>();

foreach (var scenario in input.CommandScenarios)
    results.Add(await RunCommandScenarioAsync(scenario));
foreach (var scenario in input.ViewScenarios)
    results.Add(await RunViewScenarioAsync(scenario));

await File.WriteAllTextAsync(args[1], JsonSerializer.Serialize(results));

static async Task<ScenarioResultDto> RunCommandScenarioAsync(CommandScenarioInput scenario)
{
    try
    {
        await using var store = await SqliteEventStore.OpenAsync(":memory:");
        var registry = new DeciderRegistry(store);
        RegisterDecider(registry, scenario.DeciderTypeName, scenario.Aggregate);

        if (scenario.Given.Count > 0)
        {
            var seed = scenario.Given.Select(g => new NewEvent(g.Type, g.Data)).ToList();
            await store.AppendAsync(scenario.Aggregate, scenario.StreamId, 0, seed);
        }

        try
        {
            var produced = await registry.HandleAsync(scenario.Aggregate, scenario.StreamId, new Command(scenario.CommandName, scenario.CommandPayload));

            if (scenario.Kind == "error")
                return Result(scenario, passed: false, $"expected a refusal but the command was accepted and produced {produced.Count} event(s)");

            var got = produced.Select(e => e.Type).ToList();
            var want = scenario.ExpectedEventTypes;
            if (!got.SequenceEqual(want))
                return Result(scenario, passed: false, $"expected [{string.Join(", ", want)}], got [{string.Join(", ", got)}]");

            return Result(scenario, passed: true, $"would append {string.Join(", ", got)}");
        }
        catch (Exception ex)
        {
            if (scenario.Kind == "error")
                return Result(scenario, passed: true, "refused: " + ex.Message);
            return Result(scenario, passed: false, "the decider refused the command: " + ex.Message);
        }
    }
    catch (Exception ex)
    {
        return Result(scenario, passed: false, "verification error: " + ex.Message);
    }
}

static async Task<ScenarioResultDto> RunViewScenarioAsync(ViewScenarioInput scenario)
{
    try
    {
        await using var readModel = await SqliteReadModelStore.OpenAsync(":memory:");
        var projectionType = Type.GetType(scenario.ProjectionTypeName)
            ?? throw new InvalidOperationException($"projection type not found: {scenario.ProjectionTypeName}");
        var projection = (IProjection)Activator.CreateInstance(projectionType, readModel)!;
        await CallInitAsync(projection);

        // A scoped query param resolves through a DIFFERENT read model (the semi-join
        // side) -- its own table needs to exist and be seeded in this same in-memory
        // database before the scoped SELECT can run against it. A view scenario's
        // `given` is already cross-cutting fixture history (see ScenarioVerifier's own
        // comment), so the very same events seed the via-projection too, whichever
        // stream they actually belong to.
        var viaProjections = new Dictionary<string, IProjection>();
        foreach (var scope in scenario.Scopes)
        {
            if (viaProjections.ContainsKey(scope.ViaProjectionTypeName)) continue;
            var viaType = Type.GetType(scope.ViaProjectionTypeName)
                ?? throw new InvalidOperationException($"scope projection type not found: {scope.ViaProjectionTypeName}");
            var viaProjection = (IProjection)Activator.CreateInstance(viaType, readModel)!;
            await CallInitAsync(viaProjection);
            viaProjections[scope.ViaProjectionTypeName] = viaProjection;
        }

        // Each given event carries its OWN row key per target projection (resolved by
        // ScenarioVerifier -- see ViewGivenEventInput's doc comment), rather than one
        // key shared by the whole scenario: an unscoped multi-row view scenario
        // deliberately seeds several different entities and expects several different
        // rows out.
        long position = 0;
        foreach (var g in scenario.Given)
        {
            position++;
            var mainEv = new Event(position, $"scenario-{position}", scenario.Aggregate, g.MainKey, position, g.Type, g.Data, "{}", "1970-01-01T00:00:00.000Z");
            await projection.ApplyAsync(mainEv, CancellationToken.None);
            foreach (var (viaTypeName, viaProjection) in viaProjections)
            {
                var viaKey = g.ViaKeys.GetValueOrDefault(viaTypeName, g.MainKey);
                var viaEv = new Event(position, $"scenario-{position}", scenario.Aggregate, viaKey, position, g.Type, g.Data, "{}", "1970-01-01T00:00:00.000Z");
                await viaProjection.ApplyAsync(viaEv, CancellationToken.None);
            }
        }

        var rows = await SelectRowsAsync(readModel.Connection, scenario.TableName, scenario.QueryParams, scenario.Scopes, scenario.Filters);

        // Two `then.result` shapes are supported. A single-property object whose value
        // is a JSON array (e.g. `{"entries": [...]}`) is a NAMED-LIST result: the query
        // may return any number of rows, and each element of the array must match
        // exactly one of them, with none left over on either side -- this is the shape
        // every real scenario in practice uses (a view is a list of rows, even when
        // there's only one). Anything else is a flat single-row result (the original,
        // pre-Finding-3 shape, still used by this generator's own regression tests):
        // the query must match EXACTLY one row, compared directly.
        var (isList, expectedRows) = TryUnwrapExpectedList(scenario.ExpectedResult);
        if (isList)
        {
            var diff = ListDiff(expectedRows!, rows);
            return diff is null
                ? ViewResult(scenario, passed: true, $"the projected rows match ({rows.Count})")
                : ViewResult(scenario, passed: false, "result: " + diff);
        }

        if (rows.Count != 1)
            return ViewResult(scenario, passed: false, rows.Count == 0 ? "no row matched the query" : $"the query is ambiguous: {rows.Count} rows matched");

        var flatDiff = SubsetDiff(scenario.ExpectedResult, rows[0]);
        return flatDiff is null
            ? ViewResult(scenario, passed: true, "the projected row matches")
            : ViewResult(scenario, passed: false, "result: " + flatDiff);
    }
    catch (Exception ex)
    {
        return ViewResult(scenario, passed: false, "verification error: " + ex.Message);
    }
}

static void RegisterDecider(DeciderRegistry registry, string typeName, string aggregate)
{
    var deciderType = Type.GetType(typeName) ?? throw new InvalidOperationException($"decider type not found: {typeName}");
    var createMethod = deciderType.GetMethod("Create", BindingFlags.Public | BindingFlags.Static)
        ?? throw new InvalidOperationException($"{typeName}.Create() not found");
    var decider = createMethod.Invoke(null, null)!;
    var stateType = decider.GetType().GetGenericArguments()[0];
    var registerMethod = typeof(DeciderRegistry).GetMethods().Single(m => m.Name == "Register" && m.IsGenericMethodDefinition)
        .MakeGenericMethod(stateType);
    registerMethod.Invoke(registry, [aggregate, decider]);
}

static async Task CallInitAsync(object projection)
{
    var method = projection.GetType().GetMethod("InitAsync");
    if (method is null) return;
    await (Task)method.Invoke(projection, [CancellationToken.None])!;
}

static async Task<List<Dictionary<string, object?>>> SelectRowsAsync(
    DbConnection db, string table, string? queryParamsJson, List<ViewScopeInput> scopes, List<ViewFilterInput> filters)
{
    string sql;
    var parameters = new Dictionary<string, object?>();
    if (string.IsNullOrEmpty(queryParamsJson) || queryParamsJson == "{}")
    {
        // Unscoped: a manager-style "see everything" view. Each given event already
        // carries its own resolved row key (see ScenarioVerifier), so the rows in the
        // table already ARE exactly the fixture's distinct entities -- no WHERE needed.
        sql = $"SELECT * FROM {table}";
    }
    else
    {
        var query = JsonSerializer.Deserialize<Dictionary<string, JsonElement>>(queryParamsJson)!;
        var clauses = new List<string>();
        var i = 0;
        foreach (var (key, value) in query)
        {
            // A param this read model actually DECLARES as a dateRange filter (schema
            // 2.4.0) becomes a real `col >= @from AND col <= @to` predicate against
            // Filters[i].Field -- resolving the runtime `{kind, from?, to?}` value
            // through the SAME DateRangeResolver the real generated query route calls
            // (see its own doc comment), so the two can't independently drift on what
            // "last7Days" actually means.
            var filter = filters.Find(f => f.Param == key);
            if (filter is not null && value.ValueKind == JsonValueKind.Object)
            {
                var kind = value.GetProperty("kind").GetString()!;
                var from = value.TryGetProperty("from", out var fromEl) && fromEl.ValueKind == JsonValueKind.String ? fromEl.GetString() : null;
                var to = value.TryGetProperty("to", out var toEl) && toEl.ValueKind == JsonValueKind.String ? toEl.GetString() : null;
                var (rangeFrom, rangeTo) = DateRangeResolver.ResolveBounds(kind, from, to, DateOnly.FromDateTime(DateTime.UtcNow));
                var fromParam = $"@p{i++}";
                var toParam = $"@p{i++}";
                var column = ToSnakeCase(filter.Field);
                clauses.Add($"{column} >= {fromParam} AND {column} <= {toParam}");
                parameters[fromParam] = rangeFrom;
                parameters[toParam] = rangeTo;
                continue;
            }

            // Some queryParams (e.g. an object/array this read model declares no
            // filter for) name filtering logic no generator emits -- there's no column
            // or scope for them to become a SQL predicate. Skipping them (rather than
            // erroring) still lets the ROW CONTENT comparison below report the real gap
            // the scenario is describing, instead of a verifier crash unrelated to the
            // domain.
            if (value.ValueKind is JsonValueKind.Object or JsonValueKind.Array) continue;

            var paramName = $"@p{i++}";
            var paramValue = value.ValueKind switch
            {
                JsonValueKind.Number => (object?)value.GetDouble(),
                JsonValueKind.True or JsonValueKind.False => value.GetBoolean(),
                _ => value.GetString(),
            };
            // A scoped param names no column of THIS table at all -- the real
            // predicate is a semi-join through the via read model, matching exactly
            // what ProjectionGenerator's readModel.Scopes would emit for a real query.
            // An unscoped param is the ordinary case: it names one of this table's own
            // columns directly.
            var scope = scopes.Find(s => s.Param == key);
            clauses.Add(scope is null
                ? $"{ToSnakeCase(key)} = {paramName}"
                : $"{scope.FilterLocalColumn} IN (SELECT {scope.SelectColumn} FROM {scope.ViaTableName} WHERE {scope.MatchParamToColumn} = {paramName})");
            parameters[paramName] = paramValue;
        }
        sql = clauses.Count == 0 ? $"SELECT * FROM {table}" : $"SELECT * FROM {table} WHERE {string.Join(" AND ", clauses)}";
    }

    await using var command = db.CreateCommand();
    command.CommandText = sql;
    foreach (var (name, value) in parameters)
        command.AddParam(name, value);

    var rows = new List<Dictionary<string, object?>>();
    await using (var reader = await command.ExecuteReaderAsync())
    {
        while (await reader.ReadAsync())
        {
            var row = new Dictionary<string, object?>();
            for (var i = 0; i < reader.FieldCount; i++)
                row[reader.GetName(i)] = reader.IsDBNull(i) ? null : reader.GetValue(i);
            rows.Add(row);
        }
    }
    return rows;
}

/// <summary>Detects the named-list `then.result` shape: an object with exactly one
/// property whose value is a JSON array. Every real scenario in the model uses this
/// shape (a view is a list of rows); this generator's own flat-object regression tests
/// predate it and must keep working unchanged, which is why this is a narrow structural
/// check rather than a required convention.</summary>
static (bool IsList, List<JsonElement>? Rows) TryUnwrapExpectedList(string expectedJson)
{
    var expected = JsonSerializer.Deserialize<Dictionary<string, JsonElement>>(expectedJson);
    if (expected is { Count: 1 } && expected.First().Value.ValueKind == JsonValueKind.Array)
        return (true, expected.First().Value.EnumerateArray().ToList());
    return (false, null);
}

/// <summary>Multiset comparison for a named-list result: every expected row must
/// subset-match a distinct actual row (order-independent), and no actual row may be
/// left over -- an extra row would mean an unscoped/scoped query returned something the
/// scenario didn't ask for, which is as real a bug as a missing one.</summary>
static string? ListDiff(List<JsonElement> expectedRows, List<Dictionary<string, object?>> actualRows)
{
    var unmatched = new List<Dictionary<string, object?>>(actualRows);
    var problems = new List<string>();
    foreach (var expected in expectedRows)
    {
        var expectedJson = expected.GetRawText();
        var matchIndex = unmatched.FindIndex(a => SubsetDiff(expectedJson, a) is null);
        if (matchIndex < 0)
        {
            problems.Add($"no row matched {expectedJson}");
            continue;
        }
        unmatched.RemoveAt(matchIndex);
    }
    if (unmatched.Count > 0)
        problems.Add($"{unmatched.Count} extra row(s) not expected by any element of the result list");
    return problems.Count > 0 ? string.Join("; ", problems) : null;
}

// A subset comparison, not equality: a scenario states the fields it cares about,
// and the generated code legitimately carries more. Requiring equality would fail
// every scenario for a reason that isn't about the domain.
static string? SubsetDiff(string expectedJson, Dictionary<string, object?> actual)
{
    var expected = JsonSerializer.Deserialize<Dictionary<string, JsonElement>>(expectedJson);
    if (expected is null) return "expected value is not an object";

    var diffs = new List<string>();
    foreach (var (key, want) in expected.OrderBy(kv => kv.Key, StringComparer.Ordinal))
    {
        var column = ToSnakeCase(key);
        if (!actual.TryGetValue(column, out var got))
        {
            diffs.Add($"{key} missing");
            continue;
        }
        if (!ValuesEqual(got, want))
            diffs.Add($"{key} = {got}, expected {want}");
    }
    return diffs.Count > 0 ? string.Join("; ", diffs) : null;
}

static bool ValuesEqual(object? actual, JsonElement expected)
{
    // A groupBy field's own column (schema 2.3.0) is a JSON array of objects, stored as
    // raw TEXT -- `expected.GetString()` below would throw for an Array/Object
    // JsonElement, so this shape needs its own structural comparison rather than falling
    // into the scalar text comparison every other field type uses.
    if (expected.ValueKind is JsonValueKind.Array or JsonValueKind.Object)
    {
        if (actual is not string actualJson) return false;
        JsonElement actualParsed;
        try { actualParsed = JsonSerializer.Deserialize<JsonElement>(actualJson); }
        catch (JsonException) { return false; }
        return JsonStructurallyEqual(actualParsed, expected);
    }

    // SQLite has no native boolean storage class -- a bool column (a plain-copied
    // field, or a toggle derivation's literal 1/0) comes back from the reader as a
    // long, not a bool, so `expected: true/false` must be compared against 0/nonzero
    // rather than by exact text -- otherwise "1" == "true" always fails, which would
    // make a toggle derivation's own verify scenario always fail regardless of whether
    // the generated code is correct.
    if (expected.ValueKind is JsonValueKind.True or JsonValueKind.False)
    {
        var expectedBool = expected.ValueKind == JsonValueKind.True;
        var actualBool = actual switch
        {
            null => false,
            bool b => b,
            long l => l != 0,
            double d => d != 0,
            _ => string.Equals(actual.ToString(), "true", StringComparison.OrdinalIgnoreCase),
        };
        return actualBool == expectedBool;
    }

    var actualText = actual switch
    {
        null => "null",
        bool b => b ? "true" : "false",
        long l => l.ToString(),
        double d => d.ToString("G17"),
        _ => actual.ToString(),
    };
    var expectedText = expected.ValueKind switch
    {
        JsonValueKind.Number => expected.GetDouble().ToString("G17"),
        JsonValueKind.Null => "null",
        _ => expected.GetString(),
    };
    return actualText == expectedText;
}

/// <summary>Deep-equal for a groupBy field's nested JSON, order-independent for arrays
/// (same reasoning as <c>ListDiff</c>'s top-level rows: a grouped rollup is a list of
/// rows, and nothing about the model promises what order contributing events land the
/// entries in) and snake_case-normalized for object keys, since <paramref name="actual"/>
/// (read straight off the generated column) carries the projection's own snake_case
/// property names while <paramref name="expected"/> (the scenario's own JSON) uses the
/// model's camelCase field names -- the same normalization <c>SubsetDiff</c> already does
/// for top-level columns.</summary>
static bool JsonStructurallyEqual(JsonElement actual, JsonElement expected)
{
    switch (expected.ValueKind)
    {
        case JsonValueKind.Object:
            if (actual.ValueKind != JsonValueKind.Object) return false;
            var actualProps = actual.EnumerateObject().ToDictionary(p => p.Name, p => p.Value);
            foreach (var prop in expected.EnumerateObject())
            {
                if (!actualProps.TryGetValue(ToSnakeCase(prop.Name), out var actualValue)) return false;
                if (!JsonStructurallyEqual(actualValue, prop.Value)) return false;
            }
            return true;
        case JsonValueKind.Array:
            if (actual.ValueKind != JsonValueKind.Array) return false;
            var actualItems = actual.EnumerateArray().ToList();
            var expectedItems = expected.EnumerateArray().ToList();
            if (actualItems.Count != expectedItems.Count) return false;
            var unmatched = new List<JsonElement>(actualItems);
            foreach (var expectedItem in expectedItems)
            {
                var matchIndex = unmatched.FindIndex(a => JsonStructurallyEqual(a, expectedItem));
                if (matchIndex < 0) return false;
                unmatched.RemoveAt(matchIndex);
            }
            return true;
        case JsonValueKind.True or JsonValueKind.False:
            return actual.ValueKind == expected.ValueKind;
        case JsonValueKind.Number:
            return actual.ValueKind == JsonValueKind.Number && actual.GetDouble() == expected.GetDouble();
        case JsonValueKind.Null:
            return actual.ValueKind == JsonValueKind.Null;
        default:
            return actual.ValueKind == JsonValueKind.String && actual.GetString() == expected.GetString();
    }
}

static string ToSnakeCase(string name)
{
    var b = new System.Text.StringBuilder();
    for (var i = 0; i < name.Length; i++)
    {
        var c = name[i];
        if (char.IsUpper(c))
        {
            if (i > 0) b.Append('_');
            b.Append(char.ToLowerInvariant(c));
        }
        else
        {
            b.Append(c);
        }
    }
    return b.ToString();
}

// Local functions do NOT support overloading in C# (unlike ordinary methods) --
// two local functions named Result with different parameter types are a compile
// error ("already defined in this scope"), not a valid overload set. Named
// separately instead.
static ScenarioResultDto Result(CommandScenarioInput s, bool passed, string detail) =>
    new(s.SliceId, s.ScenarioId, s.Name, s.Kind, passed, Skipped: false, detail);

static ScenarioResultDto ViewResult(ViewScenarioInput s, bool passed, string detail) =>
    new(s.SliceId, s.ScenarioId, s.Name, "stateView", passed, Skipped: false, detail);

record HarnessInput(List<CommandScenarioInput> CommandScenarios, List<ViewScenarioInput> ViewScenarios);
record GivenEventInput(string Type, string Data);
record ViewGivenEventInput(string Type, string Data, string MainKey, Dictionary<string, string> ViaKeys);
record CommandScenarioInput(string SliceId, string ScenarioId, string Name, string Kind, string DeciderTypeName,
    string Aggregate, string StreamId, List<GivenEventInput> Given, string CommandName, string CommandPayload, List<string> ExpectedEventTypes);
record ViewScenarioInput(string SliceId, string ScenarioId, string Name, string ProjectionTypeName, string Aggregate,
    string TableName, List<ViewGivenEventInput> Given, string? QueryParams, string ExpectedResult,
    List<ViewScopeInput> Scopes, List<ViewFilterInput> Filters);
record ViewScopeInput(string Param, string ViaProjectionTypeName, string ViaTableName, string MatchParamToColumn, string SelectColumn, string FilterLocalColumn);
record ViewFilterInput(string Param, string Field, string Kind);
record ScenarioResultDto(string SliceId, string ScenarioId, string Name, string Kind, bool Passed, bool Skipped, string Detail);
